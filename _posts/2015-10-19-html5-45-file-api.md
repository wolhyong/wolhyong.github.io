## 현재 위치 한 번 가져오기

```javascript
if ('geolocation' in navigator) {
  navigator.geolocation.getCurrentPosition(
    // 성공 콜백
    (position) => {
      const { latitude, longitude, accuracy, altitude, speed } = position.coords;
      const timestamp = position.timestamp;

      console.log(`위도: ${latitude}`);
      console.log(`경도: ${longitude}`);
      console.log(`정확도: ${accuracy}미터`);
    },
    // 오류 콜백
    (error) => {
      handleGeolocationError(error);
    },
    // 옵션
    {
      enableHighAccuracy: true,  // GPS 사용 (배터리 소모 증가)
      timeout: 10000,            // 10초 타임아웃
      maximumAge: 60000          // 1분 이내 캐시된 위치 허용
    }
  );
} else {
  console.log('이 브라우저는 Geolocation을 지원하지 않습니다.');
}
```

## 오류 처리

```javascript
function handleGeolocationError(error) {
  switch (error.code) {
    case GeolocationPositionError.PERMISSION_DENIED:
      // code: 1
      alert('위치 권한이 거부됐습니다. 브라우저 설정에서 위치 권한을 허용해주세요.');
      break;

    case GeolocationPositionError.POSITION_UNAVAILABLE:
      // code: 2
      alert('현재 위치를 가져올 수 없습니다. 잠시 후 다시 시도해주세요.');
      break;

    case GeolocationPositionError.TIMEOUT:
      // code: 3
      alert('위치 정보 요청이 시간 초과됐습니다.');
      break;

    default:
      alert('알 수 없는 오류가 발생했습니다.');
  }
}
```

## watchPosition — 위치 실시간 추적

위치가 변할 때마다 콜백을 호출합니다.

```javascript
let watchId;

function startTracking() {
  watchId = navigator.geolocation.watchPosition(
    (position) => {
      const { latitude, longitude } = position.coords;
      updateMap(latitude, longitude);
    },
    handleGeolocationError,
    { enableHighAccuracy: true, maximumAge: 0 }
  );
}

function stopTracking() {
  if (watchId !== undefined) {
    navigator.geolocation.clearWatch(watchId);
    watchId = undefined;
  }
}
```

## Promise 래퍼

콜백 기반 API를 Promise로 감싸서 async/await로 사용합니다.

```javascript
function getCurrentPosition(options = {}) {
  return new Promise((resolve, reject) => {
    if (!navigator.geolocation) {
      reject(new Error('Geolocation을 지원하지 않는 브라우저입니다.'));
      return;
    }
    navigator.geolocation.getCurrentPosition(resolve, reject, {
      enableHighAccuracy: false,
      timeout: 10000,
      maximumAge: 300000, // 5분 캐시
      ...options
    });
  });
}

// 사용
async function showMyLocation() {
  try {
    const position = await getCurrentPosition();
    const { latitude, longitude } = position.coords;
    console.log(`현재 위치: ${latitude}, ${longitude}`);
    renderMap(latitude, longitude);
  } catch (error) {
    handleGeolocationError(error);
  }
}
```

## 카카오맵 연동

```html
<!DOCTYPE html>
<html lang="ko">
<head>
  <meta charset="UTF-8">
  <title>현재 위치 지도</title>
  <script src="//dapi.kakao.com/v2/maps/sdk.js?appkey=YOUR_APP_KEY"></script>
  <style>
    #map { width: 100%; height: 500px; }
  </style>
</head>
<body>
  <button id="locate-btn">내 위치 보기</button>
  <div id="map"></div>

  <script>
    let map;
    let marker;

    // 지도 초기화 (서울 중심)
    const mapContainer = document.getElementById('map');
    map = new kakao.maps.Map(mapContainer, {
      center: new kakao.maps.LatLng(37.5665, 126.9780),
      level: 5
    });

    document.getElementById('locate-btn').addEventListener('click', async () => {
      if (!navigator.geolocation) return alert('Geolocation 미지원');

      try {
        const pos = await getCurrentPosition({ enableHighAccuracy: true });
        const { latitude: lat, longitude: lng } = pos.coords;
        const position = new kakao.maps.LatLng(lat, lng);

        // 기존 마커 제거
        if (marker) marker.setMap(null);

        // 새 마커 추가
        marker = new kakao.maps.Marker({ map, position });

        // 지도 이동
        map.panTo(position);

        // 정보창
        const infowindow = new kakao.maps.InfoWindow({
          content: '<div style="padding:8px">현재 위치</div>'
        });
        infowindow.open(map, marker);

      } catch (error) {
        handleGeolocationError(error);
      }
    });

    function getCurrentPosition(opts) {
      return new Promise((resolve, reject) => {
        navigator.geolocation.getCurrentPosition(resolve, reject, opts);
      });
    }

    function handleGeolocationError(e) {
      const msgs = {
        1: '위치 권한이 거부됐습니다.',
        2: '위치를 가져올 수 없습니다.',
        3: '요청 시간이 초과됐습니다.'
      };
      alert(msgs[e.code] || '오류 발생');
    }
  </script>
</body>
</html>
```

## 위치 정보 접근 시 UX 주의사항

- **먼저 왜 필요한지 설명하세요.** 갑자기 권한 요청이 뜨면 거부율이 높아집니다. "내 주변 맛집을 찾으려면 위치 권한이 필요합니다" 처럼 맥락을 먼저 제공합니다.
- **권한 거부 시 대안을 제공하세요.** 수동 주소 입력, 기본 위치(도시 수준) 사용 등 대체 방법을 제공합니다.
- **enableHighAccuracy는 필요할 때만** 사용합니다. 배터리 소모가 크고 위치 확인 시간이 길어집니다.

## 마치며
- `getCurrentPosition` — 현재 위치 한 번 가져오기
- `watchPosition` — 위치 변화 지속 추적 (이동 중인 경우)
- `clearWatch(id)` — 추적 중지
- 오류 코드: 1(권한 거부), 2(위치 불가), 3(타임아웃)
- 권한 요청 전 사용 목적을 먼저 안내하는 것이 UX 모범 사례

## 자주 하는 실수와 주의점

초보자들이 Geolocation API을 다룰 때 가장 흔히 하는 실수는 닫는 태그를 빼먹거나 속성값을 따옴표로 감싸지 않는 것입니다. HTML은 오류에 관대한 편이지만, 이런 실수는 레이아웃이 깨지거나 스타일이 정상적으로 적용되지 않는 원인이 됩니다. 반드시 모든 태그가 올바르게 열고 닫혔는지, 속성값은 따옴표로 감쌌는지 확인하는 습관을 들이는 것이 좋습니다.

## 브라우저 동작 방식

브라우저는 HTML 문서를 위에서 아래로 읽으며 DOM(Document Object Model) 트리를 생성합니다. Geolocation API 요소가 DOM 트리에서 어떻게 해석되고 배치되는지 이해하면 더 체계적인 마크업을 작성할 수 있습니다. 개발자 도구(F12)의 Elements 패널을 열어 실제 DOM 구조를 확인하는 습관을 추천합니다.
